Задачи с простыми операторами

(14 - 4 <= 10) || (6 / 2 != 3)
14 - 4 = 10, 10 <= 10 → true
6 / 2 = 3, 3 != 3 → false
true || false → true
Ответ: true

(9 + 1 == 10) && (8 % 2 == 0)
9 + 1 = 10, 10 == 10 → true
8 % 2 = 0, 0 == 0 → true
true && true → true
Ответ: true

(7 - 2 < 5) || (4 * 3 != 12)
7 - 2 = 5, 5 < 5 → false
4 * 3 = 12, 12 != 12 → false
false || false → false
Ответ: false

(18 / 6 >= 3) && (5 + 5 == 10)
18 / 6 = 3, 3 >= 3 → true
5 + 5 = 10, 10 == 10 → true
true && true → true
Ответ: true

(11 % 3 != 2) || (9 - 3 == 6)
11 % 3 = 2, 2 != 2 → false
9 - 3 = 6, 6 == 6 → true
false || true → true
Ответ: true

(5 * 2 < 11) && (14 / 2 == 7)
5 * 2 = 10, 10 < 11 → true
14 / 2 = 7, 7 == 7 → true
true && true → true
Ответ: true

(8 + 2 <= 10) || (7 % 2 != 1)
8 + 2 = 10, 10 <= 10 → true
7 % 2 = 1, 1 != 1 → false
true || false → true
Ответ: true

(16 / 4 > 3) && (3 + 4 == 7)
16 / 4 = 4, 4 > 3 → true
3 + 4 = 7, 7 == 7 → true
true && true → true
Ответ: true

(10 % 2 == 0) || (5 - 3 != 1)
10 % 2 = 0, 0 == 0 → true
5 - 3 = 2, 2 != 1 → true
true || true → true
Ответ: true

Задачи с приоритетом операторов

7 - 2 * (2 + 3) / 5 + 3 * 2
Скобки: 2 + 3 = 5
Умножения и деления слева направо:
2 * 5 = 10
10 / 5 = 2
3 * 2 = 6
Теперь сложения и вычитания: 7 - 2 + 6 = 5 + 6 = 11
Ответ: 11

7 - 2 * 2 + 3 / 3 + 5 * 2
Умножения и деления слева направо:
2 * 2 = 4
3 / 3 = 1
5 * 2 = 10
Сложения и вычитания слева направо: 7 - 4 + 1 + 10 = 3 + 1 + 10 = 14
Ответ: 14

((4 * 3) < 13) && !(8 / 4 == 2) || (10 % 3 != 1)
Арифметика и сравнения:
4 * 3 = 12, 12 < 13 → true
8 / 4 = 2, 2 == 2 → true, !true → false
10 % 3 = 1, 1 != 1 → false
Логика: true && false || false → сначала &&: false, потом ||: false || false = false
Ответ: false

(((14 - 5) > 8) || (9 / 3 == 3)) && !(7 + 0 != 7)
Скобки и арифметика:
14 - 5 = 9, 9 > 8 → true
9 / 3 = 3, 3 == 3 → true
true || true → true
7 + 0 = 7, 7 != 7 → false, !false → true
Итоговое: true && true = true
Ответ: true

((2 + 3) <= 5) && (16 / 4 == 4) || !(6 % 2 == 0)
Скобки и арифметика:
2 + 3 = 5, 5 <= 5 → true
16 / 4 = 4, 4 == 4 → true
6 % 2 = 0, 0 == 0 → true, !true → false
Логика: true && true || false → сначала &&: true, потом ||: true || false = true
Ответ: true

Задачи с инкрементом и декрементом

(var f = 6; --f >= 5) || ((3 * 2 == 6) && (9 - 4 != 5))
var f = 6, затем --f → f становится 5, 5 >= 5 → true
Правая часть: 3 * 2 = 6, 6 == 6 → true; 9 - 4 = 5, 5 != 5 → false; true && false → false
Итог: true || false → true
Ответ: true

(var g = 0; g++ != 0) && !((4 + 2 == 6) || (5 / 1 == 5))
var g = 0, затем g++ (постфиксный инкремент): сначала используем старое значение 0, потом увеличиваем. 0 != 0 → false
Правая часть: 4 + 2 = 6, 6 == 6 → true; 5 / 1 = 5, 5 == 5 → true; true || true → true; !true → false
Итог: false && false → false (даже если бы правая часть была true, левая уже false)
Ответ: false

!(var h = 7; h-- > 6) || (8 + 1 == 9) && (3 % 2 != 1)
var h = 7, затем h-- (постфиксный декремент): сначала используем 7, потом уменьшаем. 7 > 6 → true, затем !true → false
Средняя часть: 8 + 1 = 9, 9 == 9 → true
Правая часть: 3 % 2 = 1, 1 != 1 → false
Логика: приоритет && выше ||, поэтому сначала считаем true && false = false, потом false || false = false
Ответ: false

((var i = 5; i++ == 5) || (2 * 2 != 4)) && (10 / 2 == 5)
var i = 5, затем i++ (постфикс): сначала сравниваем старое значение 5 == 5 → true, потом увеличиваем
Вторая часть в скобках: 2 * 2 = 4, 4 != 4 → false
Левые скобки: true || false → true
Правая часть: 10 / 2 = 5, 5 == 5 → true
Итог: true && true → true
Ответ: true

(var j = 4; j-- < 4) && (6 + 0 == 6) || !(5 % 2 != 1)
var j = 4, затем j-- (постфикс): сначала используем 4, потом уменьшаем. 4 < 4 → false
Средняя часть: 6 + 0 = 6, 6 == 6 → true
Правая часть: 5 % 2 = 1, 1 != 1 → false, !false → true
Сначала &&: false && true = false
Потом ||: false || true = true
Ответ: true
